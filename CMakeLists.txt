cmake_minimum_required(VERSION 3.25)
project(Crackhood VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Crackhood requires Windows x64")
endif()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /utf-8 /permissive-)
include(CTest)
add_library(crackhood_core STATIC src/core/path.cpp src/core/cache.cpp src/ipc/protocol.cpp src/broker/profiles.cpp src/shell/pidl.cpp)
target_include_directories(crackhood_core PUBLIC src)
add_library(crackhood_ipc STATIC src/ipc/win_pipe.cpp src/ipc/client.cpp src/ipc/changes.cpp src/shell/launch.cpp)
target_link_libraries(crackhood_ipc PUBLIC crackhood_core advapi32 ole32 shell32)
add_library(CrackhoodShell SHARED src/shell/dll.cpp src/shell/folder.cpp src/shell/data_object.cpp src/shell/drop_target.cpp src/shell/context_menu.cpp src/shell/Crackhood.def)
target_sources(CrackhoodShell PRIVATE src/shell/resources.rc)
target_link_libraries(CrackhoodShell PRIVATE crackhood_core crackhood_ipc shell32 shlwapi ole32 oleaut32 propsys)
add_library(crackhood_sdk STATIC src/broker/sdk_transport.cpp src/broker/xbdm_stream.cpp src/broker/jobs.cpp)
target_include_directories(crackhood_sdk PRIVATE "$ENV{XEDK}/include/win32")
target_link_libraries(crackhood_sdk PUBLIC crackhood_core ws2_32)
add_executable(crackhood_broker src/broker/main.cpp)
target_sources(crackhood_broker PRIVATE src/broker/shell_job.cpp)
target_link_libraries(crackhood_broker PRIVATE crackhood_sdk crackhood_ipc)
add_executable(crackhood_ui WIN32 src/ui/main.cpp)
target_sources(crackhood_ui PRIVATE src/ui/resources.rc)
target_link_libraries(crackhood_ui PRIVATE crackhood_ipc comctl32 shell32 ole32)
add_executable(crackhood_tests tests/test_main.cpp tests/path_tests.cpp tests/pidl_tests.cpp tests/shell_tests.cpp tests/cache_tests.cpp tests/ipc_tests.cpp src/shell/folder.cpp)
target_sources(crackhood_tests PRIVATE src/shell/resources.rc)
target_link_libraries(crackhood_tests PRIVATE crackhood_core crackhood_ipc shell32 shlwapi ole32 oleaut32 propsys)
target_sources(crackhood_tests PRIVATE tests/sdk_error_tests.cpp)
target_sources(crackhood_tests PRIVATE tests/job_tests.cpp)
target_sources(crackhood_tests PRIVATE tests/stream_tests.cpp)
target_sources(CrackhoodShell PRIVATE src/shell/staging.cpp)
target_sources(crackhood_tests PRIVATE src/shell/staging.cpp)
target_sources(crackhood_tests PRIVATE tests/data_object_tests.cpp src/shell/data_object.cpp src/shell/drop_target.cpp src/shell/context_menu.cpp)
target_sources(crackhood_tests PRIVATE tests/profile_tests.cpp)
target_sources(crackhood_tests PRIVATE tests/context_menu_tests.cpp)
target_sources(crackhood_tests PRIVATE tests/shell_job_tests.cpp src/broker/shell_job.cpp)
target_link_libraries(crackhood_tests PRIVATE crackhood_sdk)
add_test(NAME core_tests COMMAND crackhood_tests)
add_test(NAME ui_layout COMMAND crackhood_ui --verify-layout)
add_executable(crackhood_probe tools/probe.cpp)
add_executable(crackhood_shell_smoke tools/shell_smoke.cpp)
add_executable(crackhood_benchmark tools/benchmark.cpp)
add_executable(crackhood_neighborhood_audit tools/neighborhood_audit.cpp)
add_executable(crackhood_progress_probe tools/progress_probe.cpp)
target_link_libraries(crackhood_progress_probe PRIVATE shell32 ole32)
target_include_directories(crackhood_neighborhood_audit PRIVATE src)
target_link_libraries(crackhood_neighborhood_audit PRIVATE shell32 ole32)
target_link_libraries(crackhood_benchmark PRIVATE crackhood_ipc)
target_include_directories(crackhood_shell_smoke PRIVATE src)
target_link_libraries(crackhood_shell_smoke PRIVATE shell32 ole32)
target_include_directories(crackhood_probe PRIVATE "$ENV{XEDK}/include/win32")
add_test(NAME probe_missing_sdk COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/probe_missing_sdk.ps1" -Probe $<TARGET_FILE:crackhood_probe>)
add_custom_target(crackhood_installer
  COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-installer.ps1" -Configuration $<CONFIG>
  DEPENDS CrackhoodShell crackhood_broker crackhood_ui
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
